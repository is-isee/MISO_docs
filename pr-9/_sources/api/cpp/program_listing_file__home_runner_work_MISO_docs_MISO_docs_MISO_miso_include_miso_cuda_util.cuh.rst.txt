   inline void check_error(cudaError_t code, const char *file, int line,
                           bool abort = true) {
     if (code != cudaSuccess) {
       const char *errstr = cudaGetErrorString(code);
       std::fprintf(stderr, "CUDA Error: %s %s %d\n", errstr, file, line);
       std::fflush(stderr);
       if (abort)
         MPI_Abort(mpi::comm(), EXIT_FAILURE);
     }
   }
